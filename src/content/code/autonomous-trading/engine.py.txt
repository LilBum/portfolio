"""Per-symbol scan: fetch data, build a plan, gate on risk, optionally execute.

Trimmed from the live engine - logging, fallback, and reconciliation paths
have been elided to keep the control flow readable.
"""
import uuid

from .engines import VwapPullbackSignalEngine
from .risk import RiskEngine
from .planner import Planner
from .data.provider_factory import create_market_data_provider


class PlannerApp:
    def __init__(self, config: dict) -> None:
        self.config = config
        self.provider, self.provider_name = create_market_data_provider(config)
        signal_engine = VwapPullbackSignalEngine(config["strategy"])
        self.planner = Planner(signal_engine, RiskEngine(), config["options"])
        self.execution_enabled = bool(config.get("execution", {}).get("enabled", False))
        self._live_mode = config.get("mode", "paper").lower() == "live"

    def run(self) -> list:
        plans, run_id = [], str(uuid.uuid4())

        # Live trading is pinned to the production broker; never run live on a
        # fallback/free data feed.
        if self._live_mode and self.provider_name != "webull":
            raise RuntimeError(f"Live mode requires the production provider, got {self.provider_name}")

        for symbol in self.config["strategy"]["symbols"]:
            try:
                bars = self.provider.get_intraday_bars(
                    symbol,
                    self.config["strategy"]["history_period"],
                    self.config["strategy"]["interval"],
                )
                chain = self.provider.get_options_chain(symbol, self.config["options"]["target_dte"])
            except Exception as exc:
                self.journal.log_event("error", {"symbol": symbol, "error": str(exc)})
                continue

            # The planner runs signal -> instrument selection -> risk sizing and
            # returns a fully-formed, auditable plan (ALLOWED / REJECTED + reasons).
            plan, signal, selection, risk = self.planner.build_plan(symbol, bars, chain, self.config)
            plan.run_id = run_id
            plans.append(plan)

            if self.execution_enabled and plan.status == "ALLOWED":
                self._submit_if_enabled(plan, run_id)

        return plans
