--!strict
-- First-purchase offer: prompts the dev-product once, after 10 spins or 15 minutes in session.
local Players = game:GetService("Players")
local MarketplaceService = game:GetService("MarketplaceService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Remotes = require(ReplicatedStorage.Remotes)
local TelemetryEvents = require(ReplicatedStorage.Shared.Data.TelemetryEvents)
local PlayerDataService = require(script.Parent.Parent.PlayerDataService)
local TelemetryService = require(script.Parent.Parent.TelemetryService)

local PRODUCT_ID = 0 -- TODO: real Developer Product ID
local SPIN_THRESHOLD, TIME_THRESHOLD, POLL_INTERVAL = 10, 900, 30
local PROMPT_COOLDOWN = 3

local sessionStart: { [Player]: number } = {}
local lastPromptRequest: { [Player]: number } = {}

local Offers = {}

-- show the offer once per player, on the first qualifying trigger
function Offers.fireIfEligible(player: Player, trigger: string)
	local data = PlayerDataService.GetData(player)
	if not data or data.firstPurchaseShown then
		return
	end
	data.firstPurchaseShown = true
	Remotes.FirstPurchaseOffer:FireClient(player)
	TelemetryService.fire(player, TelemetryEvents.FIRST_PURCHASE_PROMPT_SHOWN, { trigger = trigger })
end

-- spin-count trigger: offer once the museum reaches the threshold
function Offers.onSpin(player: Player, museumSize: number)
	if museumSize >= SPIN_THRESHOLD then
		Offers.fireIfEligible(player, "spin_10")
	end
end

local function promptPurchase(player: Player): boolean
	local data = PlayerDataService.GetData(player)
	if not data or not data.firstPurchaseShown then
		return false
	end
	if PRODUCT_ID <= 0 then
		return false
	end

	local now = os.clock()
	if lastPromptRequest[player] and now - lastPromptRequest[player] < PROMPT_COOLDOWN then
		return false
	end
	lastPromptRequest[player] = now

	local ok, err = pcall(MarketplaceService.PromptProductPurchase, MarketplaceService, player, PRODUCT_ID)
	if not ok then
		warn(("[Offers] PromptProductPurchase failed for %d: %s"):format(player.UserId, tostring(err)))
	end
	return ok
end

Players.PlayerAdded:Connect(function(player)
	sessionStart[player] = os.time()
end)
Players.PlayerRemoving:Connect(function(player)
	sessionStart[player] = nil
	lastPromptRequest[player] = nil
end)
for _, player in Players:GetPlayers() do
	sessionStart[player] = os.time()
end

-- time trigger: offer after 15 minutes in session
task.spawn(function()
	while true do
		task.wait(POLL_INTERVAL)
		for player, startTime in sessionStart do
			if os.time() - startTime >= TIME_THRESHOLD then
				Offers.fireIfEligible(player, "time_15min")
			end
		end
	end
end)

Remotes.RequestFirstPurchasePrompt.OnServerInvoke = promptPurchase

return Offers
