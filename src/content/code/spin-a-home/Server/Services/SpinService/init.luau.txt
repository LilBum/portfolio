--!strict
local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local Shared = ReplicatedStorage.Shared
local RarityRoller = require(Shared.Utils.RarityRoller)
local ArchetypeRoller = require(Shared.Utils.ArchetypeRoller)
local AppraisalCalc = require(Shared.Utils.AppraisalCalc)
local HouseRank = require(Shared.Utils.HouseRank)
local TemplateConfig = require(Shared.Data.TemplateConfig)
local GenieDialogueConfig = require(Shared.Data.GenieDialogueConfig)
local TelemetryEvents = require(Shared.Data.TelemetryEvents)
local Types = require(Shared.Types)

local Remotes = require(ReplicatedStorage.Remotes)
local PlayerDataService = require(script.Parent.PlayerDataService)
local PlotService = require(script.Parent.PlotService)
local QuestService = require(script.Parent.QuestService)
local GenieMoodService = require(script.Parent.GenieMoodService)
local TelemetryService = require(script.Parent.TelemetryService)
local GlobalLeaderboardService = require(script.Parent.GlobalLeaderboardService)
local OnboardingTelemetryService = require(script.Parent.OnboardingTelemetryService)
local RateLimit = require(script.RateLimit)
local MuseumCap = require(script.MuseumCap)
local Offers = require(script.Offers)

type HouseRecord = Types.HouseRecord

-- spin bonus Wishbuck ranges per rarity
local SPIN_BONUS: { [string]: { min: number, max: number } } = {
	Common = { min = 10, max = 100 },
	Uncommon = { min = 100, max = 500 },
	Rare = { min = 500, max = 2_000 },
	Epic = { min = 2_000, max = 10_000 },
	Legendary = { min = 10_000, max = 50_000 },
	Mythic = { min = 50_000, max = 500_000 },
	Secret = { min = 500_000, max = 5_000_000 },
}

local TIER = RarityRoller.RARITY_TIER
local PITY_THRESHOLD = 80 -- non-Epic spins -> guaranteed Epic+
local LAMP_POSITION, AREA_RADIUS = Vector3.new(0, 6.5, 0), 200
local rng = Random.new()

local function rollSpinBonus(rarity: string): number
	local r = SPIN_BONUS[rarity] or SPIN_BONUS.Common
	return math.floor(r.min + rng:NextNumber() * (r.max - r.min))
end

-- announce a rare roll to every player in scope
local function broadcastRare(
	player: Player,
	rarity: string,
	archetype: string,
	scope: string,
	inRange: ((Player) -> boolean)?
)
	for _, p in Players:GetPlayers() do
		if not inRange or inRange(p) then
			Remotes.RareAnnounce:FireClient(
				p,
				{ playerName = player.Name, rarity = rarity, archetype = archetype, scope = scope }
			)
		end
	end
end

local function handleSpin(player: Player): (boolean, string?)
	local data = PlayerDataService.GetData(player)
	if not data then
		return false, "data_not_loaded"
	end

	local rlOk, rlReason = RateLimit.check(player)
	if not rlOk then
		return false, rlReason
	end

	TelemetryService.fire(
		player,
		TelemetryEvents.SPIN_INITIATED,
		{ lampTier = data.lampTier, location = data.locationUnlocked, spinCount = #data.museum }
	)

	-- hard pity: force Epic+ at the threshold, else roll normally
	local rarity = if (data.pityCount or 0) >= PITY_THRESHOLD
		then RarityRoller.forceMinRank(data.lampTier, TIER.Epic)
		else RarityRoller.roll(data.lampTier)
	local rolledTier = TIER[rarity] or 0
	data.pityCount = if rolledTier >= TIER.Epic then 0 else (data.pityCount or 0) + 1

	local archetype = ArchetypeRoller.roll(data.locationUnlocked, GenieMoodService.getArchetypeBias())
	local appraisal =
		math.floor(AppraisalCalc.compute(rarity, data.locationUnlocked) * GenieMoodService.getAppraisalMult())
	local spinBonus = rollSpinBonus(rarity)
	local houseId = HttpService:GenerateGUID(false)
	data.wishbucks += spinBonus

	local templates = TemplateConfig[archetype] or TemplateConfig.Mundane or {}
	local templateId = #templates > 0 and tostring(templates[rng:NextInteger(1, #templates)]) or ""

	local incoming: HouseRecord = {
		id = houseId,
		rarity = rarity,
		archetype = archetype,
		appraisal = appraisal,
		templateId = templateId,
		spunAt = os.time(),
		lampTier = data.lampTier,
		locationAtSpin = data.locationUnlocked,
		outfitAtSpin = data.equippedOutfitId,
	}

	-- the house currently on the plot, for the outrank comparison
	local current: HouseRecord? = nil
	if data.displayHouseId then
		for _, record in data.museum do
			if record.id == data.displayHouseId then
				current = record
				break
			end
		end
	end

	table.insert(data.museum, incoming)

	-- evict over the museum cap after adding the new house, so the cap logic can pick the weakest record
	local evicted, refund = MuseumCap.evict(data)
	local keptIncoming = not evicted or evicted.id ~= incoming.id
	if evicted then
		Remotes.MuseumAutoSell:FireClient(
			player,
			{ removedRarity = evicted.rarity, removedTemplateId = evicted.templateId, refund = refund }
		)
		TelemetryService.fire(
			player,
			TelemetryEvents.SOFT_CURRENCY_EARNED,
			{ amount = refund, source = "museum_autosell" }
		)
		if current and current.id == evicted.id then
			current = nil
			data.displayHouseId = nil
		end
	end

	-- promote to the plot only if it beats the current display
	local didOutrank = keptIncoming and HouseRank.outranks(incoming, current)
	if didOutrank then
		local previousId = data.displayHouseId
		data.displayHouseId = houseId
		PlotService.onHouseChanged(player, incoming)
		Remotes.PlotChanged:FireClient(
			player,
			{ houseId = houseId, rarity = rarity, archetype = archetype, appraisal = appraisal }
		)
		TelemetryService.fire(
			player,
			TelemetryEvents.HOUSE_DISPLAYED_ON_PLOT,
			{ houseId = houseId, replacedHouseId = previousId }
		)
	end

	local quips = GenieDialogueConfig.ByRarity[rarity] or GenieDialogueConfig.ByRarity.Common
	Remotes.SpinResult:FireClient(player, {
		rarity = rarity,
		archetype = archetype,
		appraisal = appraisal,
		spinBonus = spinBonus,
		houseId = houseId,
		didOutrank = didOutrank,
		genieQuip = quips[rng:NextInteger(1, #quips)],
		pityCount = data.pityCount,
		pityThreshold = PITY_THRESHOLD,
		autoSold = not keptIncoming,
	})

	-- Mythic+ announces server-wide; Legendary only to nearby players
	if rolledTier >= TIER.Mythic then
		broadcastRare(player, rarity, archetype, "server")
	elseif rolledTier >= TIER.Legendary then
		broadcastRare(player, rarity, archetype, "area", function(p)
			local hrp = p.Character and p.Character:FindFirstChild("HumanoidRootPart") :: BasePart?
			return hrp ~= nil and (hrp.Position - LAMP_POSITION).Magnitude <= AREA_RADIUS
		end)
	end

	QuestService.onSpin(player)
	QuestService.onWishbucksEarned(player, spinBonus)

	TelemetryService.fire(player, TelemetryEvents.SOFT_CURRENCY_EARNED, { amount = spinBonus, source = "spin_bonus" })
	TelemetryService.fire(player, TelemetryEvents.MUSEUM_HOUSE_ADDED, { rarity = rarity, appraisal = appraisal })
	TelemetryService.fire(player, TelemetryEvents.SPIN_COMPLETE, {
		rarity = rarity,
		archetype = archetype,
		appraisal = appraisal,
		location = data.locationUnlocked,
		lampTier = data.lampTier,
		isNewBest = didOutrank,
		museumSize = #data.museum,
	})

	-- first-spin onboarding funnel
	if #data.museum == 1 then
		TelemetryService.fire(player, TelemetryEvents.FTUE_FIRST_SPIN, { rarity = rarity })
		TelemetryService.fire(player, TelemetryEvents.CORE_LOOP_FIRST_COMPLETE, {})
		OnboardingTelemetryService.logFirstSpin(player)
		if didOutrank then
			OnboardingTelemetryService.logFirstHouseShown(player)
		end
	end

	GlobalLeaderboardService.markDirty(player)
	Offers.onSpin(player, #data.museum)
	return true
end

Remotes.RequestSpin.OnServerInvoke = handleSpin

local SpinService = {}
return SpinService
