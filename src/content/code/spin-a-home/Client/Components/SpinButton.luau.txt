--!strict
-- Services
local TweenService = game:GetService("TweenService")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local React = require(ReplicatedStorage.Packages.React)

type Props = {
	spinning: boolean,
	disabled: boolean?,
	onSpin: () -> (),
}

-- one palette per state; the render picks `theme` and feeds it everywhere
type Theme = { top: Color3, mid: Color3, bot: Color3, stroke: Color3, inner: Color3, text: Color3, label: string }

local GOLD: Theme = {
	top = Color3.fromRGB(255, 220, 80), mid = Color3.fromRGB(255, 180, 35), bot = Color3.fromRGB(195, 120, 0),
	stroke = Color3.fromRGB(120, 70, 0), inner = Color3.fromRGB(255, 240, 200), text = Color3.fromRGB(80, 40, 0),
	label = "🪔  SPIN!",
}
local SPINNING: Theme = {
	top = Color3.fromRGB(165, 100, 255), mid = Color3.fromRGB(165, 100, 255), bot = Color3.fromRGB(96, 38, 186),
	stroke = Color3.fromRGB(60, 25, 130), inner = Color3.fromRGB(220, 195, 255), text = Color3.fromRGB(245, 235, 255),
	label = "WISHING…",
}
local LOCKED: Theme = {
	top = Color3.fromRGB(95, 82, 130), mid = Color3.fromRGB(95, 82, 130), bot = Color3.fromRGB(55, 45, 90),
	stroke = Color3.fromRGB(65, 55, 115), inner = Color3.fromRGB(120, 105, 165), text = Color3.fromRGB(155, 140, 195),
	label = "🪔  SPIN!",
}

local SHADOW_COLOR = Color3.fromRGB(0, 0, 0)
local SHIMMER_COLOR = Color3.fromRGB(255, 252, 220)
local CORNER = UDim.new(0, 18)
local BTN_W, BTN_H = 280, 76
local SHADOW_EXTRA_HEIGHT = 14

local SHIMMER_INFO = TweenInfo.new(0.45, Enum.EasingStyle.Linear, Enum.EasingDirection.Out, -1, false, 3.05)
local SHIMMER_START = UDim2.fromScale(-0.5, 0)
local SHIMMER_END = UDim2.fromScale(1.5, 0)

local function DropShadow()
	return React.createElement("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.new(0.5, 0, 0.5, 7),
		Size = UDim2.fromOffset(BTN_W - 4, BTN_H),
		BackgroundColor3 = SHADOW_COLOR, BackgroundTransparency = 0.55,
		BorderSizePixel = 0, ZIndex = 1, Active = false,
	}, { Corner = React.createElement("UICorner", { CornerRadius = CORNER }) })
end

-- glossy top-half highlight, tinted by the active theme
local function TopHighlight(color: Color3)
	return React.createElement("Frame", {
		Size = UDim2.fromScale(1, 0.45), BackgroundColor3 = color, BackgroundTransparency = 0.55,
		BorderSizePixel = 0, ZIndex = 3, Active = false,
	}, {
		Corner = React.createElement("UICorner", { CornerRadius = CORNER }),
		Gradient = React.createElement("UIGradient", { Transparency = NumberSequence.new(0.4, 1), Rotation = 90 }),
	})
end

-- sweeping shimmer; the effect below animates it via the ref
local function Shimmer(shimmerRef: { current: Frame? })
	return React.createElement("Frame", {
		Position = SHIMMER_START, Size = UDim2.fromScale(0.30, 1),
		BackgroundTransparency = 1, BorderSizePixel = 0, ZIndex = 5, ref = shimmerRef,
	}, {
		Gradient = React.createElement("UIGradient", {
			Transparency = NumberSequence.new({
				NumberSequenceKeypoint.new(0, 1),
				NumberSequenceKeypoint.new(0.5, 0.45),
				NumberSequenceKeypoint.new(1, 1),
			}),
			Rotation = 18, Color = ColorSequence.new(SHIMMER_COLOR),
		}),
	})
end

local function SpinButton(props: Props)
	local active = not props.spinning and props.disabled ~= true
	local isLocked = props.disabled == true and not props.spinning
	local theme = props.spinning and SPINNING or (isLocked and LOCKED or GOLD)

	local shimmerRef = React.useRef(nil :: Frame?)
	React.useEffect(function()
		local frame = shimmerRef.current
		if not frame or not active then return end
		frame.Position = SHIMMER_START
		local tween = TweenService:Create(frame, SHIMMER_INFO, { Position = SHIMMER_END })
		tween:Play()
		return function() tween:Cancel() end
	end, { active })

	local function handleClick()
		if props.spinning or props.disabled then return end
		props.onSpin()
	end

	return React.createElement("Frame", {
		Size = UDim2.fromOffset(BTN_W, BTN_H + SHADOW_EXTRA_HEIGHT),
		BackgroundTransparency = 1, ClipsDescendants = false, Active = false,
	}, {
		DropShadow = DropShadow(),
		MainBtn = React.createElement("TextButton", {
			AnchorPoint = Vector2.new(0.5, 0.5), Position = UDim2.fromScale(0.5, 0.5),
			Size = UDim2.fromOffset(BTN_W, BTN_H),
			BackgroundColor3 = theme.top, BorderSizePixel = 0, AutoButtonColor = false, ClipsDescendants = true,
			Text = theme.label, TextColor3 = theme.text, TextScaled = true, Font = Enum.Font.GothamBlack, ZIndex = 2,
			[React.Event.Activated] = handleClick,
		}, {
			Corner = React.createElement("UICorner", { CornerRadius = CORNER }),
			BodyGradient = React.createElement("UIGradient", {
				Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, theme.top),
					ColorSequenceKeypoint.new(0.5, theme.mid),
					ColorSequenceKeypoint.new(1, theme.bot),
				}),
				Rotation = 90,
			}),
			OuterStroke = React.createElement("UIStroke", {
				Color = theme.stroke, Thickness = 3, Transparency = active and 0 or 0.4,
				ApplyStrokeMode = Enum.ApplyStrokeMode.Border, LineJoinMode = Enum.LineJoinMode.Round,
			}),
			TopHighlight = TopHighlight(theme.inner),
			Pad = React.createElement("UIPadding", {
				PaddingLeft = UDim.new(0, 18), PaddingRight = UDim.new(0, 18),
				PaddingTop = UDim.new(0, 14), PaddingBottom = UDim.new(0, 14),
			}),
			Shimmer = active and Shimmer(shimmerRef) or nil,
		}),
	})
end

return SpinButton
