-- Tracks each player's currently-equipped pets and their summed multiplier.
-- Per-pet values come from ReplicatedStorage.Pets[name].Settings.Multiplier.
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local PetMultipliers = {
	Multipliers = {}, -- [playerName] = { [petId] = multiplier }
}

function PetMultipliers.ReadPlayerMultiplier(PlayerName)
	local total = 0
	for _, value in PetMultipliers.Multipliers[PlayerName] or {} do
		total += value
	end
	return math.max(total, 1)
end

function PetMultipliers.GetPetsEquipped(PlayerName)
	local count = 0
	for _ in PetMultipliers.Multipliers[PlayerName] or {} do
		count += 1
	end
	return count
end

function PetMultipliers.AddPlayer(PlayerName)
	PetMultipliers.Multipliers[PlayerName] = {}
end

function PetMultipliers.AddPet(PlayerName, PetInstance)
	local PetModel = ReplicatedStorage.Pets[PetInstance.PetName.Value]
	PetMultipliers.Multipliers[PlayerName] = PetMultipliers.Multipliers[PlayerName] or {}
	PetMultipliers.Multipliers[PlayerName][tonumber(PetInstance.Name)] = PetModel.Settings.Multiplier.Value
end

function PetMultipliers.RemovePet(PlayerName, PetID)
	if PetMultipliers.Multipliers[PlayerName] then
		PetMultipliers.Multipliers[PlayerName][PetID] = nil
	end
end

return PetMultipliers
