-- Global stat leaderboards backed by OrderedDataStores. Stats are log-encoded to
-- fit huge currency values into the ordered-store integer key space, then decoded
-- for display. Usernames are cached to avoid repeat GetNameFromUserIdAsync calls.
local DS = game:GetService("DataStoreService")
local RS = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")
local ServerStorage = game:GetService("ServerStorage")

local Main = {}

local Cache = Instance.new("Folder")
Cache.Name = "Cache"
Cache.Parent = ServerStorage
local UserIds = Instance.new("Folder")
UserIds.Name = "UserIds"
UserIds.Parent = Cache
local Names = Instance.new("Folder")
Names.Name = "Names"
Names.Parent = Cache

local Short = require(RS.Modules.Utilities).Short
local DataSave = RS["Game Settings"].DataSave.Value
local DATASTORE_ATTEMPTS = 3

function Main.GetNameFromUserId(Id)
	local Cached = Cache.UserIds:FindFirstChild(tostring(Id))
	if Cached then
		return tostring(Cached.Value)
	end

	local new
	pcall(function()
		new = Players:GetNameFromUserIdAsync(Id)
	end)
	if not new then
		return nil
	end

	local x = Instance.new("StringValue")
	x.Name = tostring(Id)
	x.Value = tostring(new)
	x.Parent = Cache.UserIds
	if not Cache.Names:FindFirstChild(tostring(new)) then
		local y = Instance.new("IntValue")
		y.Name = tostring(new)
		y.Value = tonumber(Id)
		y.Parent = Cache.Names
	end
	return tostring(new)
end

-- log-scale a value into a stable integer ordering key
local function EncodeNumber(Number)
	return math.round(math.log10(Number + 1) * 270000000000)
end
local function DecodeNumber(Number)
	return math.round(10 ^ (Number / 270000000000) - 1)
end

local function withRetry(label, callback)
	local lastErr
	for attempt = 1, DATASTORE_ATTEMPTS do
		local ok, result = pcall(callback)
		if ok then
			return result
		end
		lastErr = result
		warn(
			("[LeaderboardHandler] %s failed (%d/%d): %s"):format(label, attempt, DATASTORE_ATTEMPTS, tostring(result))
		)
		task.wait(0.5 * attempt)
	end
	return nil, lastErr
end

function Main.UploadLeaderboardStats(Player, Info)
	if Player:FindFirstChild("Loaded") and Player.Loaded.Value then
		local Stat = EncodeNumber(Player.Data[Info.Folder][Info.Stat].Value)
		local Datastore = DS:GetOrderedDataStore(Info.Datastore .. DataSave)
		withRetry("SetAsync", function()
			Datastore:SetAsync(Player.UserId, Stat)
		end)
		task.wait(0.05)
	end
end

function Main.UpdateLeaderboard(Leaderboard, Info)
	Leaderboard = workspace.Map.Leaderboards[Leaderboard]
	local Datastore = DS:GetOrderedDataStore(Info.Datastore .. DataSave)
	local Pages = withRetry("GetSortedAsync", function()
		return Datastore:GetSortedAsync(false, 100)
	end)
	if not Pages then
		return
	end

	local Page = Pages:GetCurrentPage()
	task.wait()

	local Sort = Leaderboard.SurfaceGui.MainFrame.Players.Scroll
	for Rank, Data in Page do
		local LBInstance = Sort:FindFirstChild(tostring(Rank))
		LBInstance.Visible = true
		LBInstance.Rank.Text = "#" .. Rank
		LBInstance.PlrName.Text = Main.GetNameFromUserId(Data.key) or "Failed to load"
		local ok, image = pcall(
			Players.GetUserThumbnailAsync,
			Players,
			Data.key,
			Enum.ThumbnailType.HeadShot,
			Enum.ThumbnailSize.Size420x420
		)
		LBInstance.PlayerIcon.Image = ok and image or ""
		LBInstance.Amount.Text = Short.en(DecodeNumber(Data.value))
	end
end

function Main.Setup(Module)
	for _, Leaderboard in workspace.Map.Leaderboards:GetChildren() do
		if not Module[Leaderboard.Name] then
			continue
		end -- leaderboard no longer defined
		local MainFrame = Leaderboard.SurfaceGui.MainFrame
		local isCurrency = Leaderboard.Name == "Currency"
		MainFrame.Title.Text = isCurrency and "Top 100 Most " .. RS["Game Settings"].CurrencyName.Value
			or Module[Leaderboard.Name].Title
		MainFrame.Players.Amount.Text = isCurrency and RS["Game Settings"].CurrencyName.Value
			or Module[Leaderboard.Name].Stat

		local Template = Leaderboard:FindFirstChild("Template") or script.Template
		for i = 1, 100 do
			local Clone = Template:Clone()
			Clone.LayoutOrder = i
			Clone.Name = i
			Clone.Rank.Text, Clone.PlrName.Text, Clone.Amount.Text = "", "", ""
			Clone.Parent = MainFrame.Players.Scroll
		end
	end
end

return Main
