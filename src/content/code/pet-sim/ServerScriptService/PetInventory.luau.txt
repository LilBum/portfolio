-- Pet inventory: equip/unequip/delete (single or batch), storage limits, and
-- rendering equipped pets into the world. Server-authoritative.
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

local Remotes = ReplicatedStorage:WaitForChild("Remotes")
local Modules = ReplicatedStorage.Modules
local Multipliers = require(Modules.Multipliers)
local PetMultipliers = require(Modules.PetMultipliers)

local LoadEquipped -- forward-declared so PlayerAdded can reference it

local function isReady(plr)
	return plr and plr:FindFirstChild("Loaded") and plr.Loaded.Value
end

Players.PlayerAdded:Connect(function(Player)
	local PetFolder = Instance.new("Folder")
	PetFolder.Name = Player.Name
	PetFolder.Parent = workspace.PlayerPets

	repeat
		task.wait()
	until (Player:FindFirstChild("Loaded") and Player.Loaded.Value) or Player.Parent == nil
	if Player.Parent == nil then
		return
	end
	LoadEquipped(Player)
end)

Players.PlayerRemoving:Connect(function(Player)
	local folder = workspace.PlayerPets:FindFirstChild(Player.Name)
	if folder then
		folder:Destroy()
	end
end)

-- spawn equipped pets that aren't in the world yet, despawn ones that were unequipped
function LoadEquipped(Player)
	local EquippedPets = PetMultipliers.Multipliers[Player.Name] or {}
	for PetId in EquippedPets do
		if not workspace.PlayerPets[Player.Name]:FindFirstChild(tostring(PetId)) then
			local PetInstance = Player.Data.Pets:FindFirstChild(tostring(PetId))
			if not PetInstance then
				continue
			end

			local PetModel = ReplicatedStorage.Pets:FindFirstChild(PetInstance.PetName.Value)
			if not PetModel then
				continue
			end

			local NewPet = PetModel:Clone()
			NewPet.Name = tostring(PetId)
			for _, Part in NewPet:GetChildren() do
				if Part:IsA("BasePart") then
					Part.Anchored = true
				end
			end

			local Billboard = script.BillboardGui:Clone()
			Billboard.PetName.Text = PetInstance.PetName.Value
			Billboard.PetRarity.Text = PetModel.Settings.Rarity.Value
			Billboard.StudsOffset = Vector3.new(0, NewPet.MainPart.Size.Y, 0)
			Billboard.Parent = NewPet

			NewPet:PivotTo(Player.Character.HumanoidRootPart.CFrame)
			NewPet.Parent = workspace.PlayerPets[Player.Name]
		end
	end
	for _, PlayerPet in workspace.PlayerPets[Player.Name]:GetChildren() do
		if not EquippedPets[tonumber(PlayerPet.Name)] then
			PlayerPet:Destroy()
		end
	end
end

local LoadEquippedEvent = Instance.new("BindableEvent")
LoadEquippedEvent.Name = "LoadEquipped"
LoadEquippedEvent.Event:Connect(LoadEquipped)
LoadEquippedEvent.Parent = script

local function equipPet(Player, Pet)
	if Pet.Equipped.Value then
		return
	end
	if Multipliers.GetMaxPetsEquipped(Player) <= Player.NonSaveValues.PetsEquipped.Value then
		return
	end -- equip cap
	Pet.Equipped.Value = true
	PetMultipliers.AddPet(Player.Name, Pet)
	Player.NonSaveValues.PetsEquipped.Value = PetMultipliers.GetPetsEquipped(Player.Name)
end

local function unequipPet(Player, Pet)
	if not Pet.Equipped.Value then
		return
	end
	Pet.Equipped.Value = false
	PetMultipliers.RemovePet(Player.Name, tonumber(Pet.Name))
	Player.NonSaveValues.PetsEquipped.Value = PetMultipliers.GetPetsEquipped(Player.Name)
end

local function equipAction(Player, Pet)
	if Pet.Equipped.Value then
		unequipPet(Player, Pet)
	else
		equipPet(Player, Pet)
	end
end

local function deleteAction(Player, Pet)
	if Pet.Equipped.Value then
		unequipPet(Player, Pet)
	end
	Pet:Destroy()
end

Remotes.Pet.OnServerEvent:Connect(function(Player, Action, Parameter)
	if not isReady(Player) then
		return
	end

	if type(Parameter) ~= "table" then -- single pet id
		local Pet = Player.Data.Pets:FindFirstChild(tostring(Parameter))
		if not Pet then
			return
		end -- nil guard
		if Action == "Equip" then
			equipAction(Player, Pet)
		elseif Action == "Delete" then
			deleteAction(Player, Pet)
		end
	else -- batch (equip-best / delete-all)
		if #Parameter > 100 then
			return
		end -- cap payload size
		for _, PetId in Parameter do
			local Pet = Player.Data.Pets:FindFirstChild(tostring(PetId))
			if Pet then
				if Action == "Equip" then
					equipAction(Player, Pet)
				elseif Action == "Delete" then
					deleteAction(Player, Pet)
				end
			end
		end
	end

	LoadEquipped(Player)
end)

-- toggle a pet's auto-delete flag (skips it on future hatches)
Remotes.AutoDelete.OnServerInvoke = function(Player, Pet)
	local entry = Player.Data.AutoDelete:FindFirstChild(tostring(Pet))
	if not entry then
		return false
	end
	entry.Value = not entry.Value
	return entry.Value
end
