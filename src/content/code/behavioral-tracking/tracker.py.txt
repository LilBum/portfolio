"""Two-object tracking with per-object Kalman filters and occlusion handling.

  - One KalmanTracker per object (target, robot). Constant-velocity model.
    State [x, y, vx, vy]; measurement [x, y].

  - Each frame: predict both filters, gate each detection against its prediction
    (reject outliers), update from the associated measurement, then detect
    occlusion - when the two predicted positions are close AND fewer than two
    detections arrive, coast on prediction rather than trust an ambiguous one.

  - Re-acquisition: a filter "lost" too long resets, so the next fresh detection
    re-seeds it instead of being gated against a stale prediction.

Stateful but owns no threads; the processing loop calls update() once per frame.
"""
from dataclasses import dataclass
from typing import Optional, Tuple

import numpy as np

from detection import Detection
from world_state import ObjectState


class KalmanTracker:
    """Constant-velocity Kalman filter for 2D position tracking.

    Implemented directly in numpy (no cv2.KalmanFilter / filterpy): small and
    transparent. State x = [px, py, vx, vy]; measurement z = [px, py].

    process_noise (Q): how much velocity can change per second - higher follows
    measurements more closely. measurement_noise (R): centroid noise - higher
    trusts predictions more and smooths harder.
    """

    def __init__(self, process_noise: float, measurement_noise: float):
        self._q = float(process_noise)
        self._r = float(measurement_noise)
        self._x = np.zeros(4, dtype=np.float64)        # [px, py, vx, vy]
        self._P = np.eye(4, dtype=np.float64) * 1e3    # large initial uncertainty
        self._H = np.array([[1, 0, 0, 0], [0, 1, 0, 0]], dtype=np.float64)
        self._R = np.eye(2, dtype=np.float64) * (self._r ** 2)
        self._initialized = False

    @property
    def initialized(self) -> bool:
        return self._initialized

    @property
    def position(self) -> Tuple[float, float]:
        return float(self._x[0]), float(self._x[1])

    @property
    def velocity(self) -> Tuple[float, float]:
        return float(self._x[2]), float(self._x[3])

    def initialize(self, x: float, y: float) -> None:
        """Seed at a position with zero velocity; tight position covariance, loose velocity."""
        self._x = np.array([x, y, 0.0, 0.0], dtype=np.float64)
        P = np.eye(4, dtype=np.float64)
        P[0, 0] = P[1, 1] = self._r ** 2
        P[2, 2] = P[3, 3] = 1e3
        self._P = P
        self._initialized = True

    def predict(self, dt: float) -> Tuple[float, float]:
        """Advance state by dt seconds. No-op until initialized."""
        if not self._initialized:
            return self.position
        F = np.array([[1, 0, dt, 0], [0, 1, 0, dt], [0, 0, 1, 0], [0, 0, 0, 1]], dtype=np.float64)
        # Standard kinematic process-noise matrix (random acceleration over dt).
        q, dt2 = self._q, dt * dt
        dt3, dt4 = dt2 * dt, dt2 * dt2
        Q = q * np.array([[dt4 / 4, 0, dt3 / 2, 0],
                          [0, dt4 / 4, 0, dt3 / 2],
                          [dt3 / 2, 0, dt2, 0],
                          [0, dt3 / 2, 0, dt2]], dtype=np.float64)
        self._x = F @ self._x
        self._P = F @ self._P @ F.T + Q
        return self.position

    def update(self, x: float, y: float) -> None:
        """Fold a position measurement into the estimate. Auto-initializes on first call."""
        if not self._initialized:
            self.initialize(x, y)
            return
        z = np.array([x, y], dtype=np.float64)
        y_innov = z - (self._H @ self._x)
        S = self._H @ self._P @ self._H.T + self._R
        K = self._P @ self._H.T @ np.linalg.inv(S)
        self._x = self._x + K @ y_innov
        self._P = (np.eye(4, dtype=np.float64) - K @ self._H) @ self._P

    def reset(self) -> None:
        self._x = np.zeros(4, dtype=np.float64)
        self._P = np.eye(4, dtype=np.float64) * 1e3
        self._initialized = False


@dataclass
class TrackerConfig:
    process_noise_target: float
    process_noise_robot: float
    measurement_noise: float
    max_frames_lost: int
    association_gate_px: float
    occlusion_distance_px: float


class TwoObjectTracker:
    """Manages target + robot Kalman trackers, occlusion logic, and identity."""

    def __init__(self, cfg: TrackerConfig):
        self._cfg = cfg
        self._target = KalmanTracker(cfg.process_noise_target, cfg.measurement_noise)
        self._robot = KalmanTracker(cfg.process_noise_robot, cfg.measurement_noise)
        self._target_misses = 0
        self._robot_misses = 0

    def update(self, target_det: Optional[Detection], robot_det: Optional[Detection],
               dt: float) -> Tuple[ObjectState, ObjectState, bool]:
        # 1) Predict both filters.
        if self._target.initialized:
            self._target.predict(dt)
        if self._robot.initialized:
            self._robot.predict(dt)

        # 2) Occlusion: both initialized, predictions close, < 2 detections.
        #    When occluded we don't trust a single detection's identity, so we let
        #    the filters keep predicting instead of updating.
        occluded = False
        if self._target.initialized and self._robot.initialized:
            tx, ty = self._target.position
            rx, ry = self._robot.position
            d = ((tx - rx) ** 2 + (ty - ry) ** 2) ** 0.5
            n_dets = (target_det is not None) + (robot_det is not None)
            occluded = d < self._cfg.occlusion_distance_px and n_dets < 2

        # 3) Associate (gate against prediction) and 4) update.
        target_meas = None if occluded else self._associate(self._target, target_det)
        robot_meas = None if occluded else self._associate(self._robot, robot_det)

        if target_meas is not None:
            self._target.update(*target_meas)
            self._target_misses = 0
        else:
            self._target_misses += 1
        if robot_meas is not None:
            self._robot.update(*robot_meas)
            self._robot_misses = 0
        else:
            self._robot_misses += 1

        # 5) Reset filters lost too long so re-acquisition isn't gated on a stale prediction.
        if self._target_misses > self._cfg.max_frames_lost:
            self._target.reset()
        if self._robot_misses > self._cfg.max_frames_lost:
            self._robot.reset()

        return (
            self._make_state(self._target, self._target_misses),
            self._make_state(self._robot, self._robot_misses),
            occluded,
        )

    def _associate(self, filt: KalmanTracker, det: Optional[Detection]) -> Optional[Tuple[float, float]]:
        """Accept a detection for this filter: any seed if uninitialized, else gate on distance."""
        if det is None:
            return None
        if not filt.initialized:
            return (det.x, det.y)
        px, py = filt.position
        if ((det.x - px) ** 2 + (det.y - py) ** 2) ** 0.5 > self._cfg.association_gate_px:
            return None  # reject as outlier
        return (det.x, det.y)

    def _make_state(self, filt: KalmanTracker, misses: int) -> ObjectState:
        if not filt.initialized:
            return ObjectState(x=0.0, y=0.0, vx=0.0, vy=0.0, status="lost", frames_since_detection=999)
        x, y = filt.position
        vx, vy = filt.velocity
        status = "detected" if misses == 0 else ("predicted" if misses <= self._cfg.max_frames_lost else "lost")
        return ObjectState(x=x, y=y, vx=vx, vy=vy, status=status, frames_since_detection=misses)
