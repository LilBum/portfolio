--!strict
-- builds the per-tier drop appearance, trail, value tag, and dropper flash

local TweenService = game:GetService("TweenService")

-- per-tier drop appearance
local TIER_VISUAL: { [number]: { size: Vector3, material: Enum.Material, color: Color3, labelPrefix: string } } = {
	[1] = { size = Vector3.new(1.2, 0.2, 0.8), material = Enum.Material.SmoothPlastic, color = Color3.fromRGB(212, 184, 150), labelPrefix = "FILE" },
	[2] = { size = Vector3.new(1.4, 0.3, 1.0), material = Enum.Material.SmoothPlastic, color = Color3.fromRGB(232, 168, 76), labelPrefix = "REPORT" },
	[3] = { size = Vector3.new(1.6, 0.4, 1.2), material = Enum.Material.Neon, color = Color3.fromRGB(74, 111, 165), labelPrefix = "★ EVIDENCE" },
}

local DropVisuals = {}

DropVisuals.TIER_VISUAL = TIER_VISUAL

-- resolve tier appearance, falling back to tier 1
function DropVisuals.tierVisual(tier: number)
	return TIER_VISUAL[tier] or TIER_VISUAL[1]
end

local function attachValueLabel(part: BasePart, prefix: string, value: number)
	local billboard = Instance.new("BillboardGui")
	billboard.Name = "ValueTag"
	billboard.Size = UDim2.new(0, 80, 0, 18)
	billboard.StudsOffset = Vector3.new(0, 0.6, 0)
	billboard.AlwaysOnTop = false
	billboard.LightInfluence = 0
	billboard.MaxDistance = 35
	billboard.Parent = part

	local label = Instance.new("TextLabel")
	label.Size = UDim2.new(1, 0, 1, 0)
	label.BackgroundTransparency = 1
	label.Font = Enum.Font.Code
	label.Text = ("%s · $%d"):format(prefix, value)
	label.TextColor3 = Color3.fromRGB(232, 224, 208)
	label.TextStrokeTransparency = 0
	label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
	label.TextScaled = true
	label.Parent = billboard
end

-- apply size/color/material/glow, value tag, and trail; kind is "chase" | "hottip" | "normal"
function DropVisuals.style(part: BasePart, tier: number, value: number, kind: string)
	local visual = DropVisuals.tierVisual(tier)
	local isChaseDrop = kind == "chase"
	local isHotTip = kind == "hottip"

	if isChaseDrop then
		-- red glow + decay timer
		part.Size = visual.size * 1.7
		part.Material = Enum.Material.Neon
		part.Color = Color3.fromRGB(220, 60, 60) -- chase red
		local light = Instance.new("PointLight")
		light.Color = Color3.fromRGB(255, 80, 80)
		light.Brightness = 4
		light.Range = 14
		light.Parent = part
		-- visible through props/walls
		local highlight = Instance.new("Highlight")
		highlight.Name = "ChaseHighlight"
		highlight.FillColor = Color3.fromRGB(255, 60, 60)
		highlight.FillTransparency = 0.55
		highlight.OutlineColor = Color3.fromRGB(255, 200, 200)
		highlight.OutlineTransparency = 0
		highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
		highlight.Parent = part
		-- sparks
		local sparks = Instance.new("ParticleEmitter")
		sparks.Texture = "rbxasset://textures/particles/sparkles_main.dds"
		sparks.Color = ColorSequence.new(Color3.fromRGB(255, 80, 80))
		sparks.Lifetime = NumberRange.new(0.4, 0.8)
		sparks.Rate = 40
		sparks.Speed = NumberRange.new(2, 4)
		sparks.SpreadAngle = Vector2.new(180, 180)
		sparks.Size = NumberSequence.new(0.3)
		sparks.Transparency = NumberSequence.new({
			NumberSequenceKeypoint.new(0, 0.2),
			NumberSequenceKeypoint.new(1, 1),
		})
		sparks.LightEmission = 0.6
		sparks.Parent = part
	elseif isHotTip then
		-- bigger, gold, neon
		part.Size = visual.size * 1.3
		part.Material = Enum.Material.Neon
		part.Color = Color3.fromRGB(255, 220, 100) -- gold
		local light = Instance.new("PointLight")
		light.Color = Color3.fromRGB(255, 220, 120)
		light.Brightness = 1.5
		light.Range = 6
		light.Parent = part
	else
		part.Size = visual.size
		part.Material = visual.material
		part.Color = visual.color
	end

	local labelPrefix = if isChaseDrop then "★ ANONYMOUS TIP — 6s ★" elseif isHotTip then "★ HOT TIP" else visual.labelPrefix
	attachValueLabel(part, labelPrefix, value)

	-- drop trail, tinted to tier color
	local trailTop = Instance.new("Attachment")
	trailTop.Name = "TrailTop"
	trailTop.Position = Vector3.new(0, part.Size.Y * 0.5, 0)
	trailTop.Parent = part
	local trailBottom = Instance.new("Attachment")
	trailBottom.Name = "TrailBottom"
	trailBottom.Position = Vector3.new(0, -part.Size.Y * 0.5, 0)
	trailBottom.Parent = part

	local trail = Instance.new("Trail")
	trail.Name = "DropTrail"
	trail.Attachment0 = trailTop
	trail.Attachment1 = trailBottom
	trail.Lifetime = if isChaseDrop then 0.55 else 0.35
	trail.MinLength = 0
	trail.LightEmission = if isChaseDrop then 0.8 elseif isHotTip then 0.6 else 0.2
	trail.Color = ColorSequence.new(
		if isChaseDrop then Color3.fromRGB(255, 80, 80) elseif isHotTip then Color3.fromRGB(255, 220, 100) else visual.color
	)
	trail.Transparency = NumberSequence.new({
		NumberSequenceKeypoint.new(0, 0.3),
		NumberSequenceKeypoint.new(1, 1),
	})
	trail.Parent = part
end

-- flash a dropper's PointLight; caller supplies the rate-limit scale
function DropVisuals.flash(flashLight: PointLight, isHotTip: boolean, rateScale: number)
	local peak = (if isHotTip then 4 else 2) * rateScale
	local up = TweenService:Create(flashLight, TweenInfo.new(0.06, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Brightness = peak })
	up:Play()
	up.Completed:Connect(function()
		TweenService:Create(flashLight, TweenInfo.new(0.18, Enum.EasingStyle.Quad, Enum.EasingDirection.Out), { Brightness = 0 }):Play()
	end)
end

return DropVisuals
