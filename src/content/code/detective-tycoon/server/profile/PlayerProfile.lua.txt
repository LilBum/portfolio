--!strict
-- pure profile data factory

-- Constants
local CURRENT_SCHEMA_VERSION = 1
local DEFAULT_RANK = "Rookie"

local PlayerProfile = {}

export type Receipt = {
	ascensionLevel: number,
	multiplier: number,
	rank: string,
	stampedAt: number,
}

-- solved-case record for the ledger; capped at 100, FIFO evict
export type SolvedCase = {
	caseTitle: string,
	solveTimeSeconds: number,
	ascensionLevel: number,
	stampedAt: number,
	weather: string?,
	wasStar: boolean?,
}

export type Profile = {
	version: number,
	permanent: {
		ascensionLevel: number,
		multiplier: number,
		rank: string,
		clockedIn: boolean,
		firstPuzzleCompleted: boolean,
		lastAscendTime: number,
		lastJoinTime: number,
		totalCashEarned: number,
		agencyName: string?,
		autoCollect: boolean,
		achievements: { [string]: boolean },
		unlockedCosmetics: { [string]: boolean },
		bestSolveTimes: { [string]: number },
		recentReceipts: { Receipt },
		-- match currency; +1 per UTC day, max 30
		matches: number,
		lastMatchClaimDate: string,
		matchBonusActive: boolean, -- consumed on next solve
		solvedCases: { SolvedCase },
	},
	currentRun: {
		cash: number,
		ownedUpgrades: { [string]: boolean },
		assignedPlotId: string?,
		cashMultiplier: number,
		streakMultiplier: number,
		streakLevel: number,
	},
}

function PlayerProfile.new(): Profile
	return {
		version = CURRENT_SCHEMA_VERSION,
		permanent = {
			ascensionLevel = 0,
			multiplier = 1,
			rank = DEFAULT_RANK,
			clockedIn = false,
			firstPuzzleCompleted = false,
			lastAscendTime = 0,
			lastJoinTime = 0,
			totalCashEarned = 0,
			agencyName = nil,
			autoCollect = false,
			achievements = {},
			unlockedCosmetics = {},
			bestSolveTimes = {},
			recentReceipts = {},
			matches = 0,
			lastMatchClaimDate = "",
			matchBonusActive = false,
			solvedCases = {},
		},
		currentRun = {
			cash = 0,
			ownedUpgrades = {},
			assignedPlotId = nil,
			cashMultiplier = 1,
			streakMultiplier = 1,
			streakLevel = 0,
		},
	}
end

return PlayerProfile
